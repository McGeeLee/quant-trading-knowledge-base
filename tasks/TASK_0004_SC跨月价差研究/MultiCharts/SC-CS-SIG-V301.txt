{ SC-CS-SIG-V301
  Optimized signal proxy for calendar-spread research.

  Data1 = SC2610 near/leader contract.
  Data2 = SC2611 far/follower contract.

  IMPORTANT:
  This Signal sends orders to Data1 only. Data2 is used for the spread signal.
  It is a single-leg proxy, not a true two-leg spread order.
  Use for signal timing research only; do not enable auto trading.
}

[IntrabarOrderGeneration = false]

Inputs:
    LookbackBars(1),
    MinLeaderMove(1.0),
    MinSpreadMove(0.0),
    MinSpread(0.0),
    MaxSpread(99999.0),
    ProfitTarget(1.0),
    StopLoss(2.0),
    MaxHoldBars(5),
    CooldownBars(2),
    MaxTradesPerDay(20),
    StartTime(915),
    EndTime(1455),
    ExcludeOpenMinute(True),
    Lots(1);

Vars:
    SpreadNow(0),
    SpreadPast(0),
    LeaderMove(0),
    SpreadMove(0),
    HoldBars(0),
    Cooldown(0),
    TradesToday(0),
    LastTradeDate(0),
    ValidBar(False),
    TimeOK(False),
    EntryOK(False);

if GetAppInfo(aiStrategyAuto) = 1 then
    RaiseRuntimeError("SC-CS-SIG-V301 is research only; auto trading disabled.");

if GetAppInfo(aiRealTimeCalc) = 1 then
    RaiseRuntimeError("SC-CS-SIG-V301 requires historical calculation.");

if Date <> LastTradeDate then
begin
    TradesToday = 0;
    LastTradeDate = Date;
end;

if Date of Data1 = Date of Data2
   and Time of Data1 = Time of Data2
   and BarStatus(1) = 2
   and BarStatus(2) = 2 then
begin

    if CurrentBar > LookbackBars then
    begin

        SpreadNow = Close of Data1 - Close of Data2;

        SpreadPast =
            Close[LookbackBars] of Data1
            - Close[LookbackBars] of Data2;

        LeaderMove =
            Close of Data1
            - Close[LookbackBars] of Data1;

        SpreadMove = SpreadNow - SpreadPast;

        TimeOK = False;

        if Time of Data1 >= StartTime
           and Time of Data1 <= EndTime then
        begin
            TimeOK = True;
        end;

        if ExcludeOpenMinute = True then
        begin
            if Time of Data1 = 901
               or Time of Data1 = 2101 then
            begin
                TimeOK = False;
            end;
        end;

        if Cooldown > 0 then
            Cooldown = Cooldown - 1;

        if MarketPosition = 0 then
        begin
            HoldBars = 0;

            if TimeOK = True
               and Cooldown = 0
               and TradesToday < MaxTradesPerDay
               and SpreadNow >= MinSpread
               and SpreadNow <= MaxSpread then
            begin
                EntryOK = False;

                { Long spread proxy: buy near contract. }
                if LeaderMove >= MinLeaderMove
                   and SpreadMove > MinSpreadMove then
                begin
                    Buy ("CS301LE") Lots contracts next bar at market;
                    EntryOK = True;
                end;

                { Short spread proxy: sell near contract. }
                if LeaderMove <= -MinLeaderMove
                   and SpreadMove < -MinSpreadMove then
                begin
                    SellShort ("CS301SE") Lots contracts next bar at market;
                    EntryOK = True;
                end;

                if EntryOK = True then
                begin
                    TradesToday = TradesToday + 1;
                    Cooldown = CooldownBars;
                    Print(
                        "SC-CS-SIG-V301 ENTRY ",
                        Date:7:0,
                        " ",
                        Time:4:0,
                        " Spread=",
                        SpreadNow:0:2,
                        " SpreadMove=",
                        SpreadMove:0:2
                    );
                end;
            end;
        end;
        
        if MarketPosition = 1 then
        begin
            HoldBars = HoldBars + 1;

            if BarsSinceEntry(0) >= MaxHoldBars then
                Sell ("CS301TIME") next bar at market;

            if PositionProfit(1) >= ProfitTarget then
                Sell ("CS301TP") next bar at market;

            if PositionProfit(1) <= -StopLoss then
                Sell ("CS301SL") next bar at market;
        end;

        if MarketPosition = -1 then
        begin
            HoldBars = HoldBars + 1;

            if BarsSinceEntry(0) >= MaxHoldBars then
                BuyToCover ("CS301TIME") next bar at market;

            if PositionProfit(1) >= ProfitTarget then
                BuyToCover ("CS301TP") next bar at market;

            if PositionProfit(1) <= -StopLoss then
                BuyToCover ("CS301SL") next bar at market;
        end;

    end;

end;
